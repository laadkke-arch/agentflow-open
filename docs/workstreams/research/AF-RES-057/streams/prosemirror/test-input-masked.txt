0010: function mkState(doc?: Node, config?: any) {
0011:   let plugins = [config ? history(config) : plugin]
0012:   if (config && config.preserveItems) plugins.push(new Plugin({historyPreserveItems: true} as any))
0013:   return EditorState.create({schema, plugins: plugins.concat(config && config.plugins || []), doc})
0014: }
0015: 
0016: function type(state: EditorState, text: string) {
0017:   return state.apply(state.tr.insertText(text))
0018: }
0019: function command(state: EditorState, command: Command) {
0020:   command(state, tr => state = state.apply(tr))
0021:   return state
0022: }
0023: 
0024: function compress(state: EditorState) {
0025:                                                                      
0026:                                   
0027:   plugin.getState(state).done = plugin.getState(state).done.compress()
0028: }
